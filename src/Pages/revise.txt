import React, { useState } from "react"

function App() {
  const [count, setCount] = useState(0)
  const increment = () => {
    if (count >= 0) {
      setCount((prv) => {
        return prv + 1
      })
    }
  }
  const decrement = () => {
    if (count > 0) {
      setCount((prev) => {
        return prev - 1
      })
    }
  }
  const reset = () => {
    setCount(0)
  }
  return (
    <div>
      <p>{count}</p>
      <button onClick={increment}>Inc</button>
      <button onClick={decrement}>Dec</button>
      <button onClick={reset}>Reset</button>
    </div>
  )
}

export default App

What is an API?
The API acts like a middleman between the frontend and the database.

What is Fetch?
JavaScript proivdes build-in method called Fetch. It is used to call an API

Why don't we write fetch() directly inside the component?
We don't write fetch directly inside the component because it will run on every render.

Q3. Why do we use useEffect([]) for API calls?
We use useEffect with an empty dependency array because it runs only once after the component is mounted. 
This prevents unnecessary API calls on every re-render."

import React,{useState,useEffect} from "react"

const API_URL=`https://jsonplaceholder.typicode.com/users`

function App(){

  const [data,setData]=useState([])
  const [isLoading,setIsLoading]=useState(false)
  const getCall=async()=>{
    setIsLoading(true)
    try{
      
      const getData=await fetch(API_URL)
      if(getData.ok){
        const result = await getData.json()
        setData(result)
      }else{
        console.log("Fetching error")
      }
    }catch(err){
      console.log(err.message)
    }finally{
      setIsLoading(false)
    }
  }
  useEffect(()=>{
    getCall()
  },[])
  if(isLoading){
    return <h1>Loading....</h1>
  }
  return(
    <>
      <div>
      <h3>User List</h3>
      {
        data.map((value)=>{
          return(
            <div key={value.id}>
              {value.name}
            </div>
          )
        })
      }
      </div>

    </>
  )
}

export default App;

| useMemo                           | useEffect                                                  |
| --------------------------------- | ---------------------------------------------------------- |
| Returns a memoized value          | Doesn't return a value                                     |
| Used for performance optimization | Used for side effects                                      |
| Caches expensive calculations     | Used for API calls, timers, subscriptions, event listeners |

useMemo
Returns a value.
Used for performance optimization.

useEffect
Doesn't return a value for rendering.
Used for side effects like:
API calls
Timers
Event listeners
Updating the document title

useMemo only when:
✅ Expensive calculations.
✅ Large array filtering.
✅ Sorting large data.
✅ Complex computations.

useCallback keeps the same function reference.
React.memo checks whether the props have changed.
Since the onClick prop is the same function reference, React.memo skips the child re-render.


import React, { useState, useCallback } from "react";

const Child = React.memo(({ onClick }) => {
  console.log("Child Render");
  return <button onClick={onClick}>Child</button>;
});

function App() {
  const [count, setCount] = useState(0);

  const handleClick = useCallback(() => {
    console.log("Clicked");
  }, []);

  return (
    <div>
      <h3>useCallback + React.memo</h3>

      <p>{count}</p>

      <button onClick={() => setCount(count + 1)}>
        Parent
      </button>

      <Child onClick={handleClick} />
    </div>
  );
}

export default App;

| Normal Function                    | useCallback                                       |
| ---------------------------------- | ------------------------------------------------- |
| Recreated on every render          | Memoized (cached)                                 |
| New function reference each render | Same function reference until dependencies change |
| No optimization                    | Used for performance optimization                 |

Props Drilling is the process of passing data from a parent component through multiple 
intermediate components to reach a deeply nested child component, 
even when the intermediate components do not use that data.

To avoid Props Drilling.
Instead of passing data through every component,
useContext allows any component to access shared data directly

useContext is used to share common data across components and avoid props drilling
import React,{useContext} from "react"
import {UserContext} from "./index.jsx"

function App(){
  const user=useContext(UserContext)
  return(
    <div>
      <h3>useContext</h3>
      <p>useContext is used to share common data across components and avoid props drilling</p>
      <b>This is come form useContext:{user.name}</b>
    </div>
  )
}

export default App

Q1. What is Redux Toolkit?
Redux Toolkit is an open-source JavaScript library that simplifies Redux development by reducing boilerplate code. It is used to manage application state in a centralized Store.


Q2. How many Stores should a Redux application have?
A Redux application should have only one Store because it acts as the central place to manage the entire application state.

Q3. What is a Slice?
A Slice is a feature-based part of the Redux Store. 
It contains the initialState, reducers, and actions for a specific feature, 
such as User, Employee, or Project.

Think:

Store
│
├── User Slice
├── Employee Slice
├── Project Slice


Q4. What is initialState?
initialState is the initial value of a Slice, not the entire Store.

Example:
userSlice

initialState:{
   name:"",
   role:""
}

Project Slice:

initialState:{
   projects:[]
}

Each Slice has its own initialState.

Q5. Who updates the Redux state?
Reducers update the Redux state based on the dispatched action.
