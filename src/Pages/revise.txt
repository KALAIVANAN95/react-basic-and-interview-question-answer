import React, { useState } from "react"

function App() {
  const [count, setCount] = useState(0)
  const increment = () => {
    if (count >= 0) {
      setCount((prv) => {
        return prv + 1
      })
    }
  }
  const decrement = () => {
    if (count > 0) {
      setCount((prev) => {
        return prev - 1
      })
    }
  }
  const reset = () => {
    setCount(0)
  }
  return (
    <div>
      <p>{count}</p>
      <button onClick={increment}>Inc</button>
      <button onClick={decrement}>Dec</button>
      <button onClick={reset}>Reset</button>
    </div>
  )
}

export default App

What is an API?
The API acts like a middleman between the frontend and the database.

What is Fetch?
JavaScript proivdes build-in method called Fetch. It is used to call an API

Why don't we write fetch() directly inside the component?
We don't write fetch directly inside the component because it will run on every render.

Q3. Why do we use useEffect([]) for API calls?
We use useEffect with an empty dependency array because it runs only once after the component is mounted. 
This prevents unnecessary API calls on every re-render."

import React,{useState,useEffect} from "react"

const API_URL=`https://jsonplaceholder.typicode.com/users`

function App(){

  const [data,setData]=useState([])
  const [isLoading,setIsLoading]=useState(false)
  const getCall=async()=>{
    setIsLoading(true)
    try{
      
      const getData=await fetch(API_URL)
      if(getData.ok){
        const result = await getData.json()
        setData(result)
      }else{
        console.log("Fetching error")
      }
    }catch(err){
      console.log(err.message)
    }finally{
      setIsLoading(false)
    }
  }
  useEffect(()=>{
    getCall()
  },[])
  if(isLoading){
    return <h1>Loading....</h1>
  }
  return(
    <>
      <div>
      <h3>User List</h3>
      {
        data.map((value)=>{
          return(
            <div key={value.id}>
              {value.name}
            </div>
          )
        })
      }
      </div>

    </>
  )
}

export default App;

import React,{useState,useMemo} from "react"

function calCulation(){
  console.log("cal")
  return 100
}

function App(){
  const [count,setCount]=useState(0)
  const [name,setName]=useState("")

  //const total=calCulation() // every time called

  const total=useMemo(()=>{
    return calCulation()
  },[])

  return(
    <div>
      <h3>Use Memo</h3>
      <div>
      <h3>Total : {total}</h3>
      <p>{count}</p>
      <button onClick={()=>setCount(count+1)}>InCrement</button>
      <hr/>
      <input value={name} onChange={(e)=>setName(e.target.value)}/>
      <h3>{name}</h3>
      </div>
    </div>
  )
}

export default App

| useMemo                           | useEffect                                                  |
| --------------------------------- | ---------------------------------------------------------- |
| Returns a memoized value          | Doesn't return a value                                     |
| Used for performance optimization | Used for side effects                                      |
| Caches expensive calculations     | Used for API calls, timers, subscriptions, event listeners |

useMemo
Returns a value.
Used for performance optimization.

useEffect
Doesn't return a value for rendering.
Used for side effects like:
API calls
Timers
Event listeners
Updating the document title

useMemo only when:
✅ Expensive calculations.
✅ Large array filtering.
✅ Sorting large data.
✅ Complex computations.

useCallback keeps the same function reference.
React.memo checks whether the props have changed.
Since the onClick prop is the same function reference, React.memo skips the child re-render.


import React, { useState, useCallback } from "react";

const Child = React.memo(({ onClick }) => {
  console.log("Child Render");
  return <button onClick={onClick}>Child</button>;
});

function App() {
  const [count, setCount] = useState(0);

  const handleClick = useCallback(() => {
    console.log("Clicked");
  }, []);

  return (
    <div>
      <h3>useCallback + React.memo</h3>

      <p>{count}</p>

      <button onClick={() => setCount(count + 1)}>
        Parent
      </button>

      <Child onClick={handleClick} />
    </div>
  );
}

export default App;
