2.Design REST APIs for Employee CRUD with pagination.
POST   /employee
GET    /employee?page=1&limit=10
GET    /employee/:id
PUT    /employee/:id
DELETE /employee/:id

{
data:[
{id:1,name:"kalai"},
{id:2,name:"Raj"}
]
page:1,
limit:10,
total:100,
totalPages:10
}
const express = require("express");
const app = express();

app.use(express.json());

const employees = [
  { id: 1, name: "Kalai", role: "Admin" },
  { id: 2, name: "Raj", role: "Manager" },
  { id: 3, name: "Vel", role: "User" }
];

// CREATE
app.post("/employees", (req, res) => {
  const newEmployee = {
    id: employees.length + 1,
    ...req.body
  };

  employees.push(newEmployee);

  res.status(201).json({
    message: "Employee Created",
    data: newEmployee
  });
});

// GET ALL + PAGINATION
app.get("/employees", (req, res) => {
  const page = Number(req.query.page) || 1;
  const limit = Number(req.query.limit) || 2;

  const start = (page - 1) * limit;
  const data = employees.slice(start, start + limit);

  res.json({
    data,
    page,
    limit,
    total: employees.length,
    totalPages: Math.ceil(employees.length / limit)
  });
});

// GET BY ID
app.get("/employees/:id", (req, res) => {
  const id = Number(req.params.id);

  const employee = employees.find(emp => emp.id === id);

  if (!employee) {
    return res.status(404).json({ message: "Employee Not Found" });
  }

  res.json(employee);
});

// UPDATE
app.put("/employees/:id", (req, res) => {
  const id = Number(req.params.id);

  const index = employees.findIndex(emp => emp.id === id);

  if (index === -1) {
    return res.status(404).json({ message: "Employee Not Found" });
  }

  employees[index] = {
    ...employees[index],
    ...req.body
  };

  res.json({
    message: "Employee Updated",
    data: employees[index]
  });
});

// DELETE
app.delete("/employees/:id", (req, res) => {
  const id = Number(req.params.id);

  const index = employees.findIndex(emp => emp.id === id);

  if (index === -1) {
    return res.status(404).json({ message: "Employee Not Found" });
  }

  employees.splice(index, 1);

  res.json({
    message: "Employee Deleted"
  });
});

app.listen(3000, () => {
  console.log("Server running on port 3000");
});

11.Write a function to return the first non-repeating character in a string.
First i count the each character using an object
then i loop the string again and return the first character whose count is one
if no unique character exits, i return null
function firstNonRepeatingChar(str) {
  let count = {};

  for (let i = 0; i < str.length; i++) {
    if (count[str[i]]) {
      count[str[i]] = count[str[i]] + 1;
    } else {
      count[str[i]] = 1;
    }
  }

  for (let i = 0; i < str.length; i++) {
    if (count[str[i]] === 1) {
      return str[i];
    }
  }

  return null;
}

13.Create a React search filter using useState.
import { useState } from "react";

function UserSearch() {
  const [search, setSearch] = useState("");

  const users = ["Kalai", "Arun", "Bala", "Kumar"];

  const filteredUsers = users.filter((user) =>
    user.toLowerCase().includes(search.toLowerCase())
  );

  return (
    <div>
      <input
        type="text"
        placeholder="Search user"
        value={search}
        onChange={(e) => setSearch(e.target.value)}
      />

      {filteredUsers.map((user) => (
        <p key={user}>{user}</p>
      ))}
    </div>
  );
}
export default UserSearch;

14.Write Express middleware to validate the Authorization header.
function validateAuthorization(req, res, next) {
  const authHeader = req.headers.authorization;

  if (!authHeader) {
    return res.status(401).json({
      message: "Authorization header is required",
    });
  }

  if (!authHeader.startsWith("Bearer ")) {
    return res.status(401).json({
      message: "Invalid authorization format",
    });
  }

  next();
}

module.exports = validateAuthorization;
This middleware checks whether the Authorization header exists and starts with Bearer . If it is invalid, it returns a 401 response. Otherwise, it calls next() to continue to the next middleware or route handler
