import React,{useState,useEffect} from "react"

function App(){

  const [input,setInput]=useState("")
  const [data,setData]=useState([])
  const [editId,setEditId]=useState(null)
  
  useEffect(()=>{
    try{
      const res=JSON.parse(localStorage.getItem("todo"))||[]
      setData(res)

    }catch(err){
      setData([])
    }
  },[])

  const handleAdd=()=>{
    if(!input){
      return alert("Fill all")
    }
    if(editId){
      const updated=data.map((value)=>value.id===editId ? {...value,input:input}:value)
      setData(updated)
      localStorage.setItem("todo",JSON.stringify(updated))
      setEditId(null)
      setInput("")
      return
    }
    const newData={id:Date.now(),input:input}
    const updated=[...data,newData]
    setData(updated)
    localStorage.setItem("todo",JSON.stringify(updated))
    setInput("")
  }
  const editIdFun=(value)=>{
    setEditId(value.id)
    setInput(value.input)
  }
  const deleteFun=(va)=>{
    const result=data?.filter((value)=>value.id !==va)
    setData(result)
    localStorage.setItem("todo",JSON.stringify(result))    
  }
  return(
    <div>
  
      <input 
      name={"input"}
      value={input}
      onChange={(e)=>setInput(e.target.value)}
      type="text"
      />
      <button onClick={handleAdd}>{editId? "Update":"Add"}</button>

      <div>
        {
          data.map((value)=>{
            return(
              <div key={value.id}>
                {value.input} <span onClick={()=>editIdFun(value)}>edit </span> <span onClick={()=>deleteFun(value.id)}>delete </span>
              </div>
            )
          })
        }
      </div>
    </div>
  )
}

export default App



import React, { useState, useEffect, useCallback } from "react";

function useFetch(url) {
  const [data, setData] = useState(null);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState(null);

  const getCall = useCallback(async () => {
    setLoading(true);
    try {
      const response = await fetch(url);
      if (!response.ok) throw new Error("Network response was not ok");
      const result = await response.json();
      setData(result);
    } catch (err) {
      setError(err.message);
    } finally {
      setLoading(false);
    }
  }, [url]);

  return { data, loading, error, getCall };
}

const API_URL = `https://jsonplaceholder.typicode.com/posts`;

function App() {
  const { data, loading, error, getCall } = useFetch(API_URL);

  useEffect(() => {
    getCall();
  }, [getCall]);

  if (loading) return <h2>Loading...</h2>;
  if (error) return <h2>Error: {error}</h2>;

  return (
    <div>
      {data?.map((value) => (
        <h1 key={value.id}>{value.title}</h1>
      ))}
    </div>
  );
}

export default App;

import React, { useState, useEffect } from "react";

const API_URL = `https://jsonplaceholder.typicode.com/users`;

function App() {
  const [data, setData] = useState([]);
  const [search, setSearch] = useState("");

  const getCall = async () => {
    try {
      const result = await fetch(API_URL);
      if (result.ok) {
        const updatedResult = await result.json();
        setData(updatedResult);
      }
    } catch (err) {
      console.log(err.message);
    }
  };

  useEffect(() => {
    getCall();
  }, []);

  const filteredData = data.filter((value) =>
    value.name.toLowerCase().includes(search.toLowerCase())
  );

  return (
    <div>
      <input
        value={search}
        onChange={(e) => setSearch(e.target.value)}
        placeholder="Search users..."
      />
      {filteredData.map((value) => (
        <div key={value.id}>
          <p>{value.name}</p>
        </div>
      ))}
    </div>
  );
}

export default App;

import React, { useState, useEffect } from "react";

const API_URL = "https://jsonplaceholder.typicode.com/users";

function App() {
  const [data, setData] = useState([]);
  const [currentPage, setCurrentPage] = useState(1);

  const itemPerPages = 3;

  const totalPages = Math.ceil(data.length / itemPerPages);

  const startIndex = (currentPage - 1) * itemPerPages;
  const endIndex = startIndex + itemPerPages;

  const currentData = data.slice(startIndex, endIndex);

  const getData = async () => {
    try {
      const datas = await fetch(API_URL);

      if (datas.ok) {
        const updated = await datas.json();
        setData(updated);
      }
    } catch (err) {
      setData([]);
    }
  };

  useEffect(() => {
    getData();
  }, []);

  return (
    <div>
      <h3>Pagination</h3>

      {currentData.map((value) => (
        <div key={value.id}>
          <h6>{value.name}</h6>
        </div>
      ))}

      <div>
        <button
          disabled={currentPage === 1}
          onClick={() => setCurrentPage(currentPage - 1)}
        >
          Prev
        </button>

        {Array.from({ length: totalPages }, (_, index) => (
          <button
            key={index}
            onClick={() => setCurrentPage(index + 1)}
          >
            {index + 1}
          </button>
        ))}

        <button
          disabled={currentPage === totalPages}
          onClick={() => setCurrentPage(currentPage + 1)}
        >
          Next
        </button>
      </div>
    </div>
  );
}

export default App;

  import React, { useState } from "react"

function App() {
  const [form, setForm] = useState({
    name: "",
    email: ""
  })
  const [err, serErr] = useState({
    name: "",
    email: ""
  })
  const handleChange = (e) => {
    const { name, value } = e.target
    setForm({
      ...form,
      [name]: value
    })
  }
  const validate = () => {
    const newError = {
      name: "",
      email: ""
    }
    if (!form.name) {
      newError.name = "Name"
    }
    if (!form.email) {
      newError.email = "email"
    }
    serErr(newError)
    return !newError.name && !newError.email;
  }
  const handleAdd = () => {
    if (!validate()) {
      return
    }
    console.log("added")
  }
  return (
    <div>

      <input name="name" value={form.name} onChange={handleChange} />
      {err.name && <span>{err.name}</span>}

      <input name="email" value={form.email} onChange={handleChange} />
      {err.email && <span>{err.email}</span>}


      <button onClick={handleAdd}>Add</button>
    </div>
  )
}

export default App
