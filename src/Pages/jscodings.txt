            first=arr[i]

function twoSumArray(arr,target){
    for(let i=0;i<arr.length;i++){
        for(let j=i+1;j<arr.length;j++){
            if(arr[i]+arr[j]===target){
                return [arr[i],arr[j]]
            }
        }
    }
}

Array: [1, 2, 4, 5]
Missing number: 3

function findMissingNumber(arr) {
  let max = arr[0];

  for (let i = 1; i < arr.length; i++) {
    if (arr[i] > max) {
      max = arr[i];
    }
  }

  for (let i = 1; i <= max; i++) {
    if (!arr.includes(i)) {
      return i;
    }
  }
}

console.log(findMissingNumber([1, 2, 4, 5]));

function firstNonRepeatingChar(str) {
  let count = {};

  // Step 1: ஒவ்வொரு character எத்தனை தடவை வந்தது என்று count பண்ணு
  for (let i = 0; i < str.length; i++) {
    if (count[str[i]]) {
      count[str[i]] = count[str[i]] + 1;
    } else {
      count[str[i]] = 1;
    }
  }

  // Step 2: first time மட்டும் வந்த character-ஐ find பண்ணு
  for (let i = 0; i < str.length; i++) {
    if (count[str[i]] === 1) {
      return str[i];
    }
  }

  return null;
}

console.log(firstNonRepeatingChar("aabbcdde"));

function groupByCateHigh(arr) {
    const results = arr.reduce((acc, item) => {
        acc[item.category] = (acc[item.category] || 0) + item.amount
        return acc
    }, {})


    let high = 0;
    let cate = ""
    for (let key in results) {
        if (results[key] > high) {
            high = results[key]

    let result = arr.reduce((acc, item) => {
        acc[item.department] = (acc[item.department] || [])
