    for(let i=str.length-1;i>=0;i--){
        result=result+str[i]
    }

        console.log("Palindrome", result)

function secondBigNo(arr){
    let first=[0]
    let second=[0]
    for(let i=0;i<arr.length;i++){
        if(arr[i]>first){
            second=first
            first=arr[i]
        }else if(arr[i]>second && arr[i]!==first){
            second=arr[i]
        }
    }
    return second
}

    let res=[]
    let count=0
    for(let i=0;i<arr.length;i++){
        if(arr[i]%2===0){
            res.push(arr[i])
            count++
        }
    }
    return { values: res, count: count }
}

function twoSumArray(arr,target){
    for(let i=0;i<arr.length;i++){
        for(let j=i+1;j<arr.length;j++){
            if(arr[i]+arr[j]===target){
                return [arr[i],arr[j]]
            }
        }
    }
}
console.log(twoSumArray([1,2,3,4,5,6],8))

    for(let i=0;i<str.length;i++){
        if(count[str[i]]){
            count[str[i]]=count[str[i]]+1

        }

}

11.Find the missing number
Array: [1, 2, 4, 5]
Missing number: 3

function findMissingNumber(arr) {
  let max = arr[0];

  for (let i = 1; i < arr.length; i++) {
    if (arr[i] > max) {
      max = arr[i];
    }
  }

  for (let i = 1; i <= max; i++) {
    if (!arr.includes(i)) {
      return i;
    }
  }
}

console.log(findMissingNumber([1, 2, 4, 5]));

function firstNonRepeatingChar(str) {
  let count = {};

  // Step 1: ஒவ்வொரு character எத்தனை தடவை வந்தது என்று count பண்ணு
  for (let i = 0; i < str.length; i++) {
    if (count[str[i]]) {
      count[str[i]] = count[str[i]] + 1;
    } else {
      count[str[i]] = 1;
    }
  }

  // Step 2: first time மட்டும் வந்த character-ஐ find பண்ணு
  for (let i = 0; i < str.length; i++) {
    if (count[str[i]] === 1) {
      return str[i];
    }
  }

  return null;
}

console.log(firstNonRepeatingChar("aabbcdde"));
